CREATE TABLE `telemetry` (
 `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
 `flight_id` varchar(64) NOT NULL,
 `pkt` varchar(8) NOT NULL,
 `mission` varchar(8) NOT NULL,
 `seq` int(10) unsigned NOT NULL,
 `ms` bigint(20) unsigned NOT NULL,
 `ts_utc` datetime DEFAULT NULL,
 `received_at` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
 `pressure_hpa` decimal(7,2) DEFAULT NULL,
 `temp_in_c` decimal(5,2) DEFAULT NULL,
 `rh_pct` decimal(5,2) DEFAULT NULL,
 `temp_out_c` decimal(5,2) DEFAULT NULL,
 `h_bme_m` decimal(7,2) DEFAULT NULL,
 `gps_lat` decimal(9,6) DEFAULT NULL,
 `gps_lon` decimal(9,6) DEFAULT NULL,
 `gps_alt_m` decimal(8,1) DEFAULT NULL,
 `rssi_dbm` smallint(6) DEFAULT NULL,
 `snr_db` decimal(4,1) DEFAULT NULL,
 `ubat_v` decimal(4,2) DEFAULT NULL,
 `ubat_percent` tinyint(3) unsigned DEFAULT NULL,
 `raw_line` varchar(512) DEFAULT NULL,
 `src_ip` varchar(45) DEFAULT NULL,
 PRIMARY KEY (`id`),
 UNIQUE KEY `uq_flight_seq` (`flight_id`,`seq`),
 KEY `ix_received_at` (`received_at`),
 KEY `ix_flight_time` (`flight_id`,`received_at`),
 KEY `ix_pkt_mission` (`pkt`,`mission`),
 KEY `ix_gps` (`gps_lat`,`gps_lon`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
